# Proxy Pool Modernization Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild the legacy proxy-pool project as a UV-managed FastAPI/Uvicorn service with Docker support, documentation, tests, and GSD planning artifacts.

**Architecture:** Convert the flat scripts into a `proxy_pool` package with separate modules for config, source parsing, checking, storage, scheduling, and API routes. Preserve `config.ini` and the existing JSON output schema while replacing the Flask subprocess loop with an in-process ASGI lifecycle scheduler.

**Tech Stack:** Python 3.12+, UV, FastAPI, Uvicorn, aiohttp, aiohttp-socks, pytest, pytest-asyncio, Docker.

```python
from dataclasses import dataclass
from typing import Literal, TypedDict

Protocol = Literal["http", "socks4", "socks5"]


@dataclass(frozen=True, slots=True)
class ProxyCandidate:
    address: str
    ip: str


@dataclass(slots=True)
class CheckedProxy:
    address: str
    country_code: str
    is_anonymous: bool
    timeout: float


class ProxyJsonItem(TypedDict):
    address: str
    countryCode: str
    is_anonymous: bool
    timeout: float


class ProxyJson(TypedDict):
    total_count_of_proxies: int
    http: list[ProxyJsonItem]
    socks4: list[ProxyJsonItem]
    socks5: list[ProxyJsonItem]


def empty_proxy_json() -> ProxyJson:
    return {
        "total_count_of_proxies": 0,
        "http": [],
        "socks4": [],
        "socks5": [],
    }


def checked_proxy_to_json(proxy: CheckedProxy) -> ProxyJsonItem:
    return {
        "address": proxy.address,
        "countryCode": proxy.country_code,
        "is_anonymous": proxy.is_anonymous,
        "timeout": proxy.timeout,
```

```python
from __future__ import annotations

from configparser import ConfigParser
from dataclasses import dataclass
from pathlib import Path

from proxy_pool.models import Protocol


@dataclass(frozen=True, slots=True)
class Settings:
    config_path: Path
    timeout: float
    max_connections: int
    sort_by_speed: bool
    save_path: Path
    json_result_filename: str
    save_format: str
    refresh_interval_seconds: int
    sources: dict[Protocol, tuple[str, ...]]

    @property
    def result_path(self) -> Path:
        return self.save_path / self.json_result_filename


def load_config(path: str | Path = "config.ini") -> Settings:
    config_path = Path(path)
    parser = ConfigParser(interpolation=None)
    read_files = parser.read(config_path, encoding="utf-8")
    if not read_files:
        raise FileNotFoundError(f"Config file not found: {config_path}")

    general = parser["General"]
    raw_save_path = general.get("SavePath", "").strip()
    save_path = Path(raw_save_path) if raw_save_path else config_path.resolve().parent

    sources: dict[Protocol, tuple[str, ...]] = {}
    for protocol, section_name in (("http", "HTTP"), ("socks4", "SOCKS4"), ("socks5", "SOCKS5")):
        section = parser[section_name]
        if not section.getboolean("Enabled", True):
            continue
        urls = tuple(line.strip() for line in section.get("Sources", "").splitlines() if line.strip())
        if urls:
            sources[protocol] = urls

    return Settings(
        config_path=config_path,
        timeout=general.getfloat("Timeout", 10),
        max_connections=general.getint("MaxConnections", 900),
        sort_by_speed=general.getboolean("SortBySpeed", True),
        save_path=save_path,
        json_result_filename=general.get("JsonResultFilename", "proxies.json"),
        save_format=general.get("SaveFormat", "json"),
        refresh_interval_seconds=general.getint("RefreshIntervalSeconds", fallback=900),
        sources=sources,
    )
```

```python
from __future__ import annotations

import asyncio
import re
from collections.abc import Mapping

from aiohttp import ClientSession

from proxy_pool.models import Protocol, ProxyCandidate

PROXY_PATTERN = re.compile(
    r"(?:^|\D)?(("
    r"(?:[1-9]|[1-9]\d|1\d{2}|2[0-4]\d|25[0-5])"
    r"\."
    r"(?:\d|[1-9]\d|1\d{2}|2[0-4]\d|25[0-5])"
    r"\."
    r"(?:\d|[1-9]\d|1\d{2}|2[0-4]\d|25[0-5])"
    r"\."
    r"(?:\d|[1-9]\d|1\d{2}|2[0-4]\d|25[0-5])"
    r"):(?:\d|[1-9]\d{1,3}|[1-5]\d{4}|6[0-4]\d{3}|65[0-4]\d{2}|655[0-2]\d|6553[0-5]))(?:\D|$)"
)


def extract_proxy_candidates(text: str) -> tuple[ProxyCandidate, ...]:
    seen: set[str] = set()
    candidates: list[ProxyCandidate] = []
    for match in PROXY_PATTERN.finditer(text):
        address = match.group(1)
        if address in seen:
            continue
        seen.add(address)
        candidates.append(ProxyCandidate(address=address, ip=address.rsplit(":", 1)[0]))
    return tuple(candidates)


async def fetch_source(session: ClientSession, url: str, timeout: float = 15) -> str:
    async with session.get(url, timeout=timeout) as response:
        return await response.text()


async def collect_candidates(sources: Mapping[Protocol, tuple[str, ...]]) -> dict[Protocol, set[ProxyCandidate]]:
    headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; rv:107.0) Gecko/20100101 Firefox/107.0"}
    results: dict[Protocol, set[ProxyCandidate]] = {protocol: set() for protocol in sources}

    async with ClientSession(headers=headers) as session:
        async def fetch_one(protocol: Protocol, url: str) -> None:
            try:
                text = await fetch_source(session, url)
            except Exception:
                return
            results[protocol].update(extract_proxy_candidates(text))

        await asyncio.gather(*(fetch_one(protocol, url) for protocol, urls in sources.items() for url in urls))

    return results
```

```python
from __future__ import annotations

import json
import os
from pathlib import Path

from proxy_pool.models import ProxyJson, empty_proxy_json


def read_proxy_json(path: Path) -> ProxyJson:
    try:
        with path.open("r", encoding="utf-8") as file:
            data = json.load(file)
    except (FileNotFoundError, json.JSONDecodeError, OSError):
        return empty_proxy_json()

    if not isinstance(data, dict):
        return empty_proxy_json()

    result = empty_proxy_json()
    for protocol in ("http", "socks4", "socks5"):
        value = data.get(protocol, [])
        result[protocol] = value if isinstance(value, list) else []
    result["total_count_of_proxies"] = sum(len(result[protocol]) for protocol in ("http", "socks4", "socks5"))
    return result


def write_proxy_json(path: Path, payload: ProxyJson) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp_path = path.with_name(f".{path.name}.{os.getpid()}.tmp")
    with tmp_path.open("w", encoding="utf-8") as file:
        json.dump(payload, file, indent=4, ensure_ascii=False)
        file.write("\n")
        file.flush()
        os.fsync(file.fileno())
    tmp_path.replace(path)
```

```python
from __future__ import annotations

import asyncio
from collections.abc import Mapping, Sequence
from time import perf_counter

from aiohttp import ClientSession
from aiohttp_socks import ProxyConnector

from proxy_pool.config import Settings, load_config
from proxy_pool.models import CheckedProxy, Protocol, ProxyCandidate, ProxyJson, checked_proxy_to_json, empty_proxy_json
from proxy_pool.sources import collect_candidates
from proxy_pool.storage import write_proxy_json

CHECK_URL = "http://ip-api.com/json/?fields=8219"


async def check_candidate(candidate: ProxyCandidate, protocol: Protocol, timeout: float, semaphore: asyncio.Semaphore) -> CheckedProxy | None:
    async with semaphore:
        proxy_url = f"{protocol}://{candidate.address}"
        start = perf_counter()
        try:
            async with ProxyConnector.from_url(proxy_url) as connector:
                async with ClientSession(connector=connector) as session:
                    async with session.get(CHECK_URL, timeout=timeout, raise_for_status=True) as response:
                        data = await response.json()
        except Exception:
            return None

    elapsed = perf_counter() - start
    query_ip = str(data.get("query", ""))
    return CheckedProxy(
        address=candidate.address,
        country_code=str(data.get("countryCode", "")),
        is_anonymous=candidate.ip != query_ip,
        timeout=elapsed,
    )


def sort_checked(proxies: Sequence[CheckedProxy], sort_by_speed: bool) -> list[CheckedProxy]:
    if sort_by_speed:
        return sorted(proxies, key=lambda proxy: proxy.timeout)
    return sorted(proxies, key=lambda proxy: tuple(int(part) for part in proxy.address.replace(":", ".").split(".")))


def build_proxy_json(proxies: Mapping[Protocol, Sequence[CheckedProxy]]) -> ProxyJson:
    payload = empty_proxy_json()
    for protocol in ("http", "socks4", "socks5"):
        payload[protocol] = [checked_proxy_to_json(proxy) for proxy in proxies.get(protocol, [])]
    payload["total_count_of_proxies"] = sum(len(payload[protocol]) for protocol in ("http", "socks4", "socks5"))
    return payload


async def run_pipeline(settings: Settings) -> ProxyJson:
    candidates = await collect_candidates(settings.sources)
    semaphore = asyncio.Semaphore(settings.max_connections)
    checked: dict[Protocol, list[CheckedProxy]] = {protocol: [] for protocol in ("http", "socks4", "socks5")}

    async def check_one(protocol: Protocol, candidate: ProxyCandidate) -> None:
        proxy = await check_candidate(candidate, protocol, settings.timeout, semaphore)
        if proxy is not None:
            checked[protocol].append(proxy)

    await asyncio.gather(*(check_one(protocol, candidate) for protocol, protocol_candidates in candidates.items() for candidate in protocol_candidates))
    sorted_checked = {protocol: sort_checked(values, settings.sort_by_speed) for protocol, values in checked.items()}
    payload = build_proxy_json(sorted_checked)
    write_proxy_json(settings.result_path, payload)
    return payload


async def async_main() -> None:
    settings = load_config()
    await run_pipeline(settings)


def main() -> None:
    asyncio.run(async_main())


if __name__ == "__main__":
    main()
```

```python
from __future__ import annotations

from pathlib import Path
from typing import Any

from fastapi import APIRouter, Query, Request

from proxy_pool.storage import read_proxy_json

router = APIRouter()


def flatten_proxy_json(payload: dict[str, Any]) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []
    for protocol in ("http", "socks4", "socks5"):
        for item in payload.get(protocol, []):
            if isinstance(item, dict):
                row = dict(item)
                row["prototype"] = protocol
                rows.append(row)
    return rows


def apply_filters(
    rows: list[dict[str, Any]],
    prototypes: str | None,
    countrycodes: str | None,
    isanonymous: str | None,
    maxtimeout: float | None,
) -> list[dict[str, Any]]:
    allowed_protocols = {value.strip().lower() for value in prototypes.split(",")} if prototypes else None
    allowed_countries = {value.strip().lower() for value in countrycodes.split(",")} if countrycodes else None
    anonymous_value = isanonymous.lower() if isanonymous is not None else None

    filtered: list[dict[str, Any]] = []
    for row in rows:
        if allowed_protocols is not None and str(row.get("prototype", "")).lower() not in allowed_protocols:
            continue
        if allowed_countries is not None and str(row.get("countryCode", "")).lower() not in allowed_countries:
            continue
        if anonymous_value is not None and str(row.get("is_anonymous", "")).lower() != anonymous_value:
            continue
        if maxtimeout is not None and float(row.get("timeout", 0)) > maxtimeout:
            continue
        filtered.append(row)
    return filtered


@router.get("/")
def list_proxies(
    request: Request,
    prototypes: str | None = None,
    countrycodes: str | None = None,
    isanonymous: str | None = None,
    maxtimeout: float | None = Query(default=None),
) -> Any:
    result_path: Path = request.app.state.result_path
    payload = read_proxy_json(result_path)
    if prototypes is None and countrycodes is None and isanonymous is None and maxtimeout is None:
        return payload
    return apply_filters(flatten_proxy_json(payload), prototypes, countrycodes, isanonymous, maxtimeout)


@router.get("/health")
def health(request: Request) -> dict[str, Any]:
    result_path: Path = request.app.state.result_path
    return {"status": "ok", "storage_path": str(result_path), "storage_exists": result_path.exists()}
```

```python
import asyncio
from contextlib import suppress

from proxy_pool.checker import run_pipeline
from proxy_pool.config import Settings


async def scheduler_loop(settings: Settings, stop_event: asyncio.Event) -> None:
    while not stop_event.is_set():
        await run_pipeline(settings)
        with suppress(asyncio.TimeoutError):
            await asyncio.wait_for(stop_event.wait(), timeout=settings.refresh_interval_seconds)
```

```python
from __future__ import annotations

import asyncio
from contextlib import asynccontextmanager, suppress
from pathlib import Path

import uvicorn
from fastapi import FastAPI

from proxy_pool.api import router
from proxy_pool.config import load_config
from proxy_pool.scheduler import scheduler_loop


@asynccontextmanager
async def lifespan(app: FastAPI):
    if app.state.start_scheduler and app.state.settings is not None:
        app.state.scheduler_stop_event = asyncio.Event()
        app.state.scheduler_task = asyncio.create_task(scheduler_loop(app.state.settings, app.state.scheduler_stop_event))
    else:
        app.state.scheduler_task = None
    try:
        yield
    finally:
        task = getattr(app.state, "scheduler_task", None)
        if task is not None:
            app.state.scheduler_stop_event.set()
            task.cancel()
            with suppress(asyncio.CancelledError):
                await task


def create_app(result_path: Path | None = None, start_scheduler: bool = True) -> FastAPI:
    app = FastAPI(title="Proxy Pool", lifespan=lifespan)
    if result_path is None:
        settings = load_config()
        app.state.settings = settings
        app.state.result_path = settings.result_path
    else:
        app.state.settings = None
        app.state.result_path = result_path
    app.state.start_scheduler = start_scheduler
    app.include_router(router)
    return app


app = create_app()


def main() -> None:
    uvicorn.run("proxy_pool.app:app", host="0.0.0.0", port=5000)


if __name__ == "__main__":
    main()
```

```markdown
# Architecture

Proxy Pool is an ASGI service built around a scrape-check-store-read loop.

## Components

- `proxy_pool.config` loads typed settings from `config.ini`.
- `proxy_pool.sources` fetches source URLs and extracts `ip:port` candidates.
- `proxy_pool.checker` validates candidates through protocol-aware proxy connectors and builds the output JSON.
- `proxy_pool.storage` atomically writes and safely reads `proxies.json`.
- `proxy_pool.scheduler` runs the refresh loop inside the ASGI process.
- `proxy_pool.api` exposes the JSON and filtering API.
- `proxy_pool.app` creates the FastAPI application and wires startup/shutdown behavior.

## Data Flow

1. Load `config.ini`.
2. Fetch source URLs for enabled protocols.
3. Extract and de-duplicate candidates.
4. Check candidates with bounded concurrency.
5. Write `proxies.json` atomically.
6. Serve stored results through the API.

## Compatibility

The output JSON shape is preserved from the legacy project. `scrap_proxies.py` and `web_server.py` remain as compatibility wrappers.
```
